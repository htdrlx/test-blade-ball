local EASE = TweenInfo.new(0.24, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
local FAST = TweenInfo.new(0.16, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

local function create(className, properties, parent)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do object[key] = value end
    object.Parent = parent
    return object
end
local function round(object, pixels)
    return create("UICorner", {CornerRadius = UDim.new(0, pixels)}, object)
end
local function outline(object, color, thickness, transparency)
    return create("UIStroke", {
        Color = color or COLORS.border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, object)
end
local function padding(object, left, right, top, bottom)
    return create("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0), PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0), PaddingBottom = UDim.new(0, bottom or 0),
    }, object)
end
local function text(parent, value, size, color, bold)
    return create("TextLabel", {
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Font = bold and Enum.Font.GothamBold or Enum.Font.GothamMedium,
        Text = value or "", TextSize = size or 15,
        TextColor3 = color or COLORS.text,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, parent)
end
local function tween(object, info, properties)
    local animation = ZTweenService:Create(object, info or EASE, properties)
    animation:Play()
    return animation
end
local function addHover(button, normal, hover)
    button.MouseEnter:Connect(function() tween(button, FAST, {BackgroundColor3 = hover}) end)
    button.MouseLeave:Connect(function() tween(button, FAST, {BackgroundColor3 = normal}) end)
end

local function makeSwitch(parent, initial, callback)
    local button = create("TextButton", {
        Size = UDim2.fromOffset(51,28), BackgroundColor3 = initial and Color3.fromRGB(215,214,222) or COLORS.switchOff,
        BorderSizePixel = 0, Text = "", AutoButtonColor = false,
    }, parent)
    round(button, 20)
    local knob = create("Frame", {
        Position = initial and UDim2.fromOffset(27,3) or UDim2.fromOffset(4,3),
        Size = UDim2.fromOffset(22,22), BorderSizePixel = 0,
        BackgroundColor3 = initial and Color3.fromRGB(35,34,40) or COLORS.switchKnob,
    }, button)
    round(knob, 20)
    local value = initial == true
    local busy = false
    local function set(nextValue, silent)
        value = nextValue == true
        tween(button, EASE, {BackgroundColor3 = value and Color3.fromRGB(215,214,222) or COLORS.switchOff})
        tween(knob, TweenInfo.new(.26, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
            Position = value and UDim2.fromOffset(27,3) or UDim2.fromOffset(4,3),
            BackgroundColor3 = value and Color3.fromRGB(35,34,40) or COLORS.switchKnob,
        })
        if not silent and callback then task.spawn(callback, value) end
    end
    button.MouseButton1Click:Connect(function()
        if busy then return end
        busy = true; set(not value); task.delay(.1, function() busy = false end)
    end)
    local switchObject = {Instance = button, Set = set, Get = function() return value end}
    switchObject.change_state = function(_, state) return set(state) end
    return switchObject
end

                end
                tween(item.Caption, PAGE, {TextColor3 = active and COLORS.white or COLORS.dim})

local ZP_FOLDER = "zenthra"

local ZP_can_write = false
pcall(function()
    if type(isfolder) == "function" and type(makefolder) == "function"
       and type(writefile) == "function" and type(readfile) == "function"
       and type(isfile) == "function" then
        if not isfolder(ZP_FOLDER) then makefolder(ZP_FOLDER) end
        if not isfolder(ZP_SUBFOLDER) then makefolder(ZP_SUBFOLDER) end
        ZP_can_write = true
    end
end)

local ZP_ready = false
local ZP_dirty = false
local ZP_fresh = true

local function ZP_save()
    if not ZP_can_write then return end
    pcall(function()
        writefile(ZP_PATH, ZP_HttpService:JSONEncode(ZCFG.saved))
    end)
end

local function ZP_load()
    if not ZP_can_write then return end
    pcall(function()
        if isfile(ZP_PATH) then
            local ok, decoded = pcall(function()
                return ZP_HttpService:JSONDecode(readfile(ZP_PATH))
            end)
            if ok and type(decoded) == "table" then
                ZCFG.saved = decoded
                for _ in pairs(decoded) do
                    ZP_fresh = false
                    break
                end
            end
        end
    end)
end

ZP_load()

local function ZP_slug(text)
    return (tostring(text or "")
        :lower()
        :gsub("[^%w]+", "_")
        :gsub("^_+", "")
        :gsub("_+$", ""))
end

local ZP_used = {}
local function ZP_key(prefix, name)
    local base = prefix ~= "" and (prefix .. "_" .. ZP_slug(name)) or ZP_slug(name)
    if base == "" then base = "widget" end
    local key, n = base, 1
    while ZP_used[key] do
        n = n + 1
        key = base .. "_" .. n
    end
    ZP_used[key] = true
    return key
end

local function ZP_mark()
    ZP_dirty = true
end

task.spawn(function()
    while task.wait(1) do
        if ZP_ready and ZP_dirty then
            ZP_dirty = false
            ZP_save()
        end
    end
end)

local ZP_restore = {}

local function ZP_off(v)
    if ZP_fresh then
        if type(v) == "boolean" then return false end
    end
    return v
end

local function ZP_persist(section, prefix)
    local raw = {
        AddToggle   = section.AddToggle,
        AddSlider   = section.AddSlider,
        AddDropdown = section.AddDropdown,
        AddInput    = section.AddInput,
    }

    function section:AddToggle(data)
        data = data or {}
        local key = ZP_key(prefix, data.Name)
        local stored = ZCFG.saved[key]
        local base = ZP_off(data.Default == true) == true
        if type(stored) == "boolean" then
            data.Default = stored
        else
            data.Default = base
            ZCFG.saved[key] = base
        end
        local user = data.Callback
        data.Callback = function(value)
            ZCFG.saved[key] = value == true
            ZP_mark()
            if user then user(value) end
        end
        local control = raw.AddToggle(self, data)
        ZP_restore[#ZP_restore + 1] = function()
            local v = ZCFG.saved[key]
            if type(v) ~= "boolean" then return end
            if v ~= (control.Get() == true) then
                control.Set(v, false)
            elseif v ~= base and user then
                task.spawn(user, v)
            end
        end
        local wrapped = { Instance = control.Instance, Get = control.Get, _key = key }
        function wrapped.Set(value, silent)
            local st = value == true
            control.Set(st, silent)
            ZCFG.saved[key] = st
            ZP_mark()
        end
        wrapped.change_state = function(_, st) return wrapped.Set(st) end
        return wrapped
    end

    function section:AddSlider(data)
        data = data or {}
        local key = ZP_key(prefix, data.Name)
        local stored = tonumber(ZCFG.saved[key])
        local base = data.Default
        if stored then
            data.Default = stored
        else
            ZCFG.saved[key] = data.Default
        end
        local user = data.Callback
        data.Callback = function(value)
            ZCFG.saved[key] = value
            ZP_mark()
            if user then user(value) end
        end
        local obj = raw.AddSlider(self, data)
        ZP_restore[#ZP_restore + 1] = function()
            local v = tonumber(ZCFG.saved[key])
            if not v then return end
            if v ~= obj.Get() then obj:Set(v, true) end
            if v ~= base and user then task.spawn(user, v) end
        end
        local rawSet = obj.Set
        function obj:Set(value, silent)
            rawSet(self, value, silent)
            ZCFG.saved[key] = self.Get()
            ZP_mark()
        end
        obj._key = key
        return obj
    end

    function section:AddDropdown(data)
        data = data or {}
        local key = ZP_key(prefix, data.Name)
        local stored = ZCFG.saved[key]
        local base = data.Default
        if type(stored) == "string" then
            data.Default = stored
        else
            ZCFG.saved[key] = data.Default
        end
        local user = data.Callback
        data.Callback = function(value)
            ZCFG.saved[key] = value
            ZP_mark()
            if user then user(value) end
        end
        local obj = raw.AddDropdown(self, data)
        ZP_restore[#ZP_restore + 1] = function()
            local v = ZCFG.saved[key]
            if type(v) ~= "string" then return end
            if v ~= obj.Get() then obj:Set(v, true) end
            if v ~= base and user then task.spawn(user, v) end
        end
        local rawSet = obj.Set
        function obj:Set(value, silent)
            rawSet(self, value, silent)
            ZCFG.saved[key] = value
            ZP_mark()
        end
        obj.update = function(_, v) return obj:Set(v) end
        obj._key = key
        return obj
    end

    function section:AddInput(data)
        data = data or {}
        local key = ZP_key(prefix, data.Name)
        local stored = ZCFG.saved[key]
        if type(stored) == "string" then
            data.Default = stored
        else
            ZCFG.saved[key] = data.Default or ""
        end
        local user = data.Callback
        data.Callback = function(value)
            ZCFG.saved[key] = value
            ZP_mark()
            if user then user(value) end
        end
        local obj = raw.AddInput(self, data)
        ZP_restore[#ZP_restore + 1] = function()
            local v = ZCFG.saved[key]
            if type(v) ~= "string" or v == "" then return end
            if v ~= obj.Get() then obj:Set(v) end
            if user then task.spawn(user, v) end
        end
        local rawSet = obj.Set
        function obj:Set(value)
            rawSet(self, value)
            ZCFG.saved[key] = tostring(value)
            ZP_mark()
        end
        obj.update = function(_, v) return obj:Set(v) end
        obj._key = key
        return obj
    end

    return section
end

        if type(stored) == "boolean" then
            config.Enabled = stored
        else
            config.Enabled = base
            ZCFG.saved[key] = base
        end
        local user = config.Callback
        config.Callback = function(state)
            ZCFG.saved[key] = state == true
            ZP_mark()
            if user then user(state) end
        end
        local section = rawAddSection(self, config)
        ZP_restore[#ZP_restore + 1] = function()
            local v = ZCFG.saved[key]
            if type(v) ~= "boolean" then return end
            if v ~= (section:GetEnabled() == true) then
                section:SetEnabled(v)
            elseif v ~= base and user then
                task.spawn(user, v)
            end
        end
        local rawSetEnabled = section.SetEnabled
        function section:SetEnabled(state, silent)
            rawSetEnabled(self, state, silent)
            ZCFG.saved[key] = state == true
            ZP_mark()
        end
        function section:change_state(state) return section:SetEnabled(state) end
        return ZP_persist(section, key)
    end
    return tab
end

local function ZP_wrap_window(window)
    local rawAddTab = window.AddTab
    function window:AddTab(name, icon)
        return ZP_wrap_tab(rawAddTab(self, name, icon))
    end
    return window
end

    can_write = function() return ZP_can_write end,
    path = ZP_PATH,
    fresh = function() return ZP_fresh end,
    saved = function() return ZCFG.saved end,
    count = function()
        local n = 0
        for _ in pairs(ZCFG.saved) do n = n + 1 end
        return n
    end,
    enabled_count = function()
        local n = 0
        for _, v in pairs(ZCFG.saved) do if v == true then n = n + 1 end end
        return n
    end,
    save_now = function() ZP_save() end,
}

local function ZP_apply()
    for _, fn in ipairs(ZP_restore) do
        pcall(fn)
    end
    ZP_ready = true
    ZP_dirty = false
end
